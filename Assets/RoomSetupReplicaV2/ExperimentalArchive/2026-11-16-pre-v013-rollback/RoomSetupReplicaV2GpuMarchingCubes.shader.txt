Shader "Genesis/RoomSetupReplicaV2GpuMarchingCubes"
{
    Properties
    {
        _BaseColor ("Color", Color) = (0.08, 1.0, 1.0, 1.0)
        _LineWidthPixels ("Line Width", Range(0.75, 4.0)) = 2.25
        _UniqueEdgesOnly ("Unique Edges Only", Float) = 0
        _StabilityProbeEnabled ("Stability Probe Enabled", Float) = 0
    }
    SubShader
    {
        Tags
        {
            "RenderType"="Transparent"
            "Queue"="Transparent+45"
            "RenderPipeline"="UniversalPipeline"
        }
        Pass
        {
            Name "RoomSetupGpuMcWire"
            Tags { "LightMode"="SRPDefaultUnlit" }
            Blend SrcAlpha OneMinusSrcAlpha
            ZWrite Off
            ZTest LEqual
            Cull Off

            HLSLPROGRAM
            #pragma vertex Vert
            #pragma fragment Frag
            #pragma target 4.5
            #pragma multi_compile_instancing

            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            StructuredBuffer<float3> _RoomSetupMcCandidateVertices;
            StructuredBuffer<float3> _RoomSetupMcFilteredVertices;
            StructuredBuffer<float> _RoomSetupMcCandidateStability;
            StructuredBuffer<uint> _RoomSetupMcCounters;
            StructuredBuffer<uint> _RoomSetupMcUniqueEdgeMasks;

            struct Attributes
            {
                uint vertexID : SV_VertexID;
                UNITY_VERTEX_INPUT_INSTANCE_ID
            };

            struct Varyings
            {
                float4 positionHCS : SV_POSITION;
                nointerpolation uint edgeEnabled : TEXCOORD0;
                float edgeCoordinate : TEXCOORD1;
                nointerpolation float stability : TEXCOORD2;
                UNITY_VERTEX_OUTPUT_STEREO
            };

            CBUFFER_START(UnityPerMaterial)
                half4 _BaseColor;
                float _LineWidthPixels;
                float _UniqueEdgesOnly;
                float _StabilityProbeEnabled;
            CBUFFER_END

            float3 LoadPosition(uint vertexID)
            {
                return _RoomSetupMcCounters[6] != 0u
                    ? _RoomSetupMcFilteredVertices[vertexID]
                    : _RoomSetupMcCandidateVertices[vertexID];
            }

            uint BuildUniqueEdgeMask(uint triangleIndex)
            {
                if (_UniqueEdgesOnly < 0.5)
                    return 7u;
                return _RoomSetupMcUniqueEdgeMasks[triangleIndex];
            }

            Varyings Vert(Attributes input)
            {
                Varyings output = (Varyings)0;
                UNITY_SETUP_INSTANCE_ID(input);
                UNITY_INITIALIZE_VERTEX_OUTPUT_STEREO(output);
                // Eighteen draw vertices become three narrow screen-space
                // quads, one for each real edge.  Only edge-local pixels are
                // rasterized; the source triangle interior is never shaded.
                uint triangleIndex = input.vertexID / 18u;
                uint triangleVertex = input.vertexID - triangleIndex * 18u;
                uint edgeIndex = triangleVertex / 6u;
                uint quadVertex = triangleVertex - edgeIndex * 6u;
                uint triangleBase = triangleIndex * 3u;
                output.edgeEnabled =
                    (BuildUniqueEdgeMask(triangleIndex) >> edgeIndex) & 1u;
                if (output.edgeEnabled == 0u)
                {
                    output.positionHCS = float4(2.0, 2.0, 1.0, 1.0);
                    output.edgeCoordinate = 0.0;
                    return output;
                }

                uint cornerA = edgeIndex;
                uint cornerB = edgeIndex == 2u ? 0u : edgeIndex + 1u;
                float3 positionA = LoadPosition(triangleBase + cornerA);
                float3 positionB = LoadPosition(triangleBase + cornerB);
                output.stability = _StabilityProbeEnabled > 0.5
                    ? min(_RoomSetupMcCandidateStability[
                              triangleBase + cornerA],
                          _RoomSetupMcCandidateStability[
                              triangleBase + cornerB])
                    : 1.0;
                float4 clipA = TransformWorldToHClip(positionA);
                float4 clipB = TransformWorldToHClip(positionB);

                float2 ndcA = clipA.xy / max(abs(clipA.w), 0.00001);
                float2 ndcB = clipB.xy / max(abs(clipB.w), 0.00001);
                float2 pixelDelta = (ndcB - ndcA) *
                    max(_ScreenParams.xy, 1.0) * 0.5;
                float pixelLength = max(length(pixelDelta), 0.0001);
                float2 normalPixels =
                    float2(-pixelDelta.y, pixelDelta.x) / pixelLength;

                // Keep the requested width and opacity in screen space at all
                // viewing distances.  The previous inverse-distance fade made
                // ordinary 1-3 m room geometry sub-pixel and nearly invisible.
                float rasterWidth = max(_LineWidthPixels, 1.0);

                bool useB = quadVertex == 2u || quadVertex == 4u ||
                            quadVertex == 5u;
                float side = quadVertex == 0u || quadVertex == 3u ||
                             quadVertex == 5u ? -1.0 : 1.0;
                output.positionHCS = useB ? clipB : clipA;
                float2 offsetNdc = normalPixels * side * rasterWidth /
                                   max(_ScreenParams.xy, 1.0);
                output.positionHCS.xy += offsetNdc * output.positionHCS.w;
                output.edgeCoordinate = side;
                return output;
            }

            half4 Frag(Varyings input) : SV_Target
            {
                UNITY_SETUP_STEREO_EYE_INDEX_POST_VERTEX(input);
                if (input.edgeEnabled == 0u)
                    discard;
                float edge = abs(input.edgeCoordinate);
                float feather = max(fwidth(edge), 0.02);
                float coverage = 1.0 - smoothstep(
                    1.0 - feather, 1.0, edge);
                clip(coverage - 0.01);
                half3 changing = half3(1.0, 0.08, 0.62);
                half3 settling = half3(1.0, 0.72, 0.05);
                half3 stabilityColor = input.stability < 0.5
                    ? lerp(changing, settling,
                           saturate(input.stability * 2.0))
                    : lerp(settling, _BaseColor.rgb,
                           saturate((input.stability - 0.5) * 2.0));
                return half4(stabilityColor,
                    _BaseColor.a * coverage);
            }
            ENDHLSL
        }
    }
}
